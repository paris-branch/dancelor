%% The default documentclasses only support font sizes up to 12pt. We use the
%% package `extsizes` and its documentclass `extbook`.
\ifsimple%
  \documentclass[14pt]{extarticle}
\else%
  \documentclass[14pt, twoside]{extarticle}
\fi%

%% Custom page geometry. It changes depending on whether the document is
%% two-sided.
\usepackage{geometry}
\geometry{
  a4paper,
  includehead,
  top=1mm,
  headheight=6mm,
  headsep=2mm,
  includefoot,
  bottom=3mm,
  footskip=6mm,
}
\ifsimple%
  \geometry{left=12mm, right=12mm} % sum should match with non-simple
\else%
  \geometry{inner=14mm, outer=10mm} % sum should match with simple
\fi%
% \geometry{showframe} %% shows frame around pages for debugging

%% Font
\usepackage{fontspec}
\defaultfontfeatures{Scale=MatchLowercase}
\setromanfont[Numbers=Uppercase]{Source Sans Pro}

%% Headers and footers
\makeatletter
\usepackage{fancyhdr}
\ifheaders%
  \fancypagestyle{default}{%
    \fancyhf{}
    \ifsimple%
      \fancyhead[L]{\leftmark}
      \fancyhead[R]{\rightmark}
      \fancyfoot[L]{\@specificity}
      \fancyfoot[R]{\thepage}
    \else%
      \fancyhead[LE]{\leftmark}
      \fancyhead[RO]{\rightmark}
      \fancyfoot[RE,LO]{\@specificity}
      \fancyfoot[LE,RO]{\thepage}
    \fi%
    \renewcommand{\headrulewidth}{0pt}
    \renewcommand{\footrulewidth}{0pt}
  }
  \fancypagestyle{norightmark}{%
    \fancyhf{}
    \ifsimple%
      \fancyhead[L]{\leftmark}
      \fancyfoot[L]{\@specificity}
      \fancyfoot[R]{\thepage}
    \else%
      \fancyhead[LE]{\leftmark}
      \fancyfoot[RE,LO]{\@specificity}
      \fancyfoot[LE,RO]{\thepage}
    \fi%
    \renewcommand{\headrulewidth}{0pt}
    \renewcommand{\footrulewidth}{0pt}
  }
  \fancypagestyle{norightmarknospecificity}{%
    \fancyhf{}
    \ifsimple%
      \fancyhead[L]{\leftmark}
      \fancyfoot[R]{\thepage}
    \else%
      \fancyhead[LE]{\leftmark}
      \fancyfoot[LE,RO]{\thepage}
    \fi%
    \renewcommand{\headrulewidth}{0pt}
    \renewcommand{\footrulewidth}{0pt}
  }
\else
  \fancypagestyle{default}{
    \fancyhf{}
    \renewcommand{\headrulewidth}{0pt}
    \renewcommand{\footrulewidth}{0pt}
  }
  \fancypagestyle{norightmark}{
    \fancyhf{}
    \renewcommand{\headrulewidth}{0pt}
    \renewcommand{\footrulewidth}{0pt}
  }
  \fancypagestyle{norightmarknospecificity}{
    \fancyhf{}
    \renewcommand{\headrulewidth}{0pt}
    \renewcommand{\footrulewidth}{0pt}
  }
\fi
\pagestyle{default}
\makeatother
